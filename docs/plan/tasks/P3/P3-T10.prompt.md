  def run_triage_job(conn, job: Job, *, adapter: Any | None = None) -> None:
      cfg = config.load()
      _raise_idle_timeout(conn, cfg)                          # design note 3 — FIRST statement
      row = _load(conn, job.subject_id)                       # explicit column list; None → PermanentError("alert not found")
      if row.status in ("closed_fp", "closed_benign", "closed_confirmed", "duplicate"):
          _log({"triage": "skipped", "alert_id": …, "status": row.status}); return None          # phase-5: never overwrite a closed alert
      if not cfg.llm_enabled:
          _set_triage_status(conn, alert_id, "unavailable"); _log({... "stopped_by": "disabled"}); return None   # no row: no prompt was sent
      correlation = domain.correlation.summarize_for_prompt(conn, _Subject(row))     # the light subject transitions.py uses
      playbook = kb.get_playbook(row.category); table = kb.get_decision_table(row.category)
      inp = TriageInput(...); facts = build_facts(inp, correlation)
      built = build_proposer_prompt(inp, facts, correlation, playbook, table, cfg=cfg)
      findings = detector.scan(built.prompt.block_index.values())
      violations = linter.lint(built.prompt.user, nonce=built.prompt.nonce, constants=builder.TEMPLATE_CONSTANTS, enums=builder.ENUM_VALUES)
      if violations: → write_run(role='proposer', result=None, gate_result={"error": "builder_violation", "violations": [...≤200 chars each]}, stopped_by='linter', …); write_event('llm.builder_violation', alert_id, 'system', payload=…); _set_triage_status(…, 'unavailable'); return None
      adapter = adapter or DeepSeekAdapter(cfg, spent_usd=partial(llm_runs.month_spend_usd, conn))
      try:
          proposal = propose(adapter, built, cfg=cfg)
      except LLMBudgetExceeded: → write_run(…, result=None, stopped_by='cap', gate_result={"error": "cap", "facts": facts}); unavailable; return
      except TransientError as exc: → _last_attempt_or_raise(conn, job, cfg, exc, built, facts)      # design note 4
      if proposal.parsed is None: → write_run(…, result=None, stopped_by='schema', gate_result={"error": "schema", "schema_errors": proposal.schema_errors, "facts": facts}, …usage/cost from proposal…); unavailable; return
      rule_check = _make_rule_check(table, facts)              # design note 5
      state = gate.run_steps_1_to_5(proposal.parsed, facts, built.prompt.block_index, repaired=proposal.repaired, rule_check=rule_check, scan=lambda _: findings)
      verdict_for_verifier = state.verdict                      # keep it: step 6 may change state.verdict, and the verifier row records what it was asked to check
      vbuilt = build_verifier_prompt(facts, table, verdict_for_verifier, state.reasons, cfg=cfg)
      (lint vbuilt too — a violation here is a template defect: raise PermanentError naming it; it cannot happen with a typed builder, and a test proves the branch exists)
      try: verification = verify(adapter, vbuilt, cfg=cfg)
      except LLMBudgetExceeded / TransientError: same two branches as above (the proposer row is NOT written on a retry — the whole transaction rolls back and the retry repeats both calls; on the last attempt both rows are written with what exists)
      gate.step6_verifier(state, verification.parsed)
      result = gate.step7_output_guard(state)
      pv = prompt_version(TRIAGE_TEMPLATE)
      g = gate.gate_result(state, prompt_version=pv, playbook_used_effective=built.playbook_used_effective)
      proposer_id = write_run(conn, run_id=uuid4(), role='proposer', subject_id=alert_id, system_prompt=built.prompt.system, user_message=built.prompt.user, result=result, gate_result=g, verifier_result=verification.parsed, injection_findings=detector.as_json(findings), citation_warnings=built.warnings, input_tokens=proposal.usage['prompt_tokens'], output_tokens=proposal.usage['completion_tokens'], latency_ms=proposal.latency_ms, model_id=proposal.model, prompt_version=pv, cost_usd=cost_usd(proposal.usage, cfg), stopped_by=None)
      write_run(conn, run_id=uuid4(), role='verifier', subject_id=alert_id, system_prompt=vbuilt.prompt.system, user_message=vbuilt.prompt.user, result=verification.parsed, gate_result={"role": "verifier", "step6": state.steps["6"], "compared_to": verdict_for_verifier, "proposer_run_id": str(proposer_id)}, verifier_result=verification.parsed, …, prompt_version=prompt_version(VERIFIER_TEMPLATE), cost_usd=cost_usd(verification.usage, cfg), stopped_by=None if verification.parsed is not None else 'schema')
      _set_triage_status(conn, alert_id, 'ready')            # UPDATE alerts SET triage_status = %s, triaged_count = triaged_count + 1 WHERE alert_id = %s — never `status`
      write_event(conn, 'triage.suggested', alert_id, 'llm', payload={"suggested_action": result["suggested_action"], "confidence": result["confidence"], "forced": g["forced"], "run_id": str(proposer_id)})

      return None

  Everything above runs inside the worker's one transaction: a retry sees no partial rows, and the `succeeded` status lands with the writes. `_set_triage_status` is the **only** `UPDATE alerts` in `tier1/`, and it names `triage_status` and `triaged_count` and nothing else (acceptance 6's grep).

  **3 · The idle timeout — the handler's first statement, and the 11-second test that proves it.** `infra/db.py`'s `transaction()` sets `idle_in_transaction_session_timeout = '10s'`; a model call is client-side idle time; a second `SET LOCAL` in the same transaction overrides the first (measured 15/09: `'2s'` then `'6s'` → `SHOW` = `6s`, session alive at 3.5 s). So: `conn.execute(sql.SQL("SET LOCAL idle_in_transaction_session_timeout = {}").format(sql.Literal(f"{cfg.JOB_LOCK_TIMEOUT_S}s")))` — 300 s by default, **the same bound after which `reclaim_stale` hands the job to another worker**, so the two limits agree by construction; measured p95 per call is 3.65 s. **Test:** a `FakeLLM` subclass whose first `complete()` sleeps **11 s**, driven through `run_forever(lambda: conn, {"triage": partial(run_triage_job, adapter=slow)}, once=True)` → the job `succeeded` and two `llm_runs` rows exist. **Red step:** comment out the `SET LOCAL` line → `psycopg.OperationalError: terminating connection due to idle-in-transaction timeout` → the job is finished `transient`, 0 rows → red. Mark the test `@pytest.mark.db` and put `# 11 s on purpose — design note 3` above it.

  **4 · The last attempt owns the outcome (planning decision 3).** `_last_attempt_or_raise(conn, job, cfg, exc, built, facts)`: if `job.attempts < cfg.JOB_MAX_ATTEMPTS` → `raise` (the worker classifies it transient and backs off; the transaction is rolled back, nothing is half-written). Else: `write_run(role='proposer', result=None, stopped_by='transient', gate_result={"error": "transient", "message": str(exc)[:200], "attempts": job.attempts, "facts": facts}, system_prompt=built…, …)`, `_set_triage_status('unavailable')`, `write_event('job.exhausted', alert_id, 'system', payload={"reason": "llm_transient", "attempts": job.attempts})`, return. The job then records `succeeded` — its contract (record an outcome) is met; `unavailable` is a valid state (phase-5). **Test:** `FakeLLM(raises=TransientError("stalled"))` with `job.attempts == cfg.JOB_MAX_ATTEMPTS` → `unavailable`, one row with `stopped_by='transient'`, a `job.exhausted` event; and with `attempts == 1` → the exception propagates and `run_forever` finishes the job `pending` with `scheduled_at` in the future, 0 rows.

  **5 · `_make_rule_check(table, facts)` — the callable step 4 gets.** Returns `(False, "no_rule_cited")` for `None`; `(False, "no_table")` when `table is None`; `(False, "table_unreviewed")` when not `table.reviewed`; `(False, "unknown_rule")` when `kb.rule_holds(...) is None`; `(False, "rule_does_not_hold")` when `False`; `(False, "rule_is_not_false_positive")` when it holds but the rule's `action != "false_positive"`; else `(True, None)`. Six strings, six tests. The `warnings` from the built prompt (`decision_table_unreviewed:<cat>` etc.) already reach `citation_warnings`; step 4's `why_not` goes to `gate_result.warnings` via P3-T08.

  **6 · Loading the row (G10).** `_load` selects exactly: `alert_id, status, triage_status, rule_id, rule_level, alert_time, severity, category, categories, occurrence_count, risk_score, description, raw_log, raw_log_truncated, agent_name, origin_host, agent_id, alert_user, srcip, mitre_ids, asset_context, identity_context, ioc_context, lookup_status, first_seen_at, last_seen_at` — 26 columns, `WHERE alert_id = %s`. `asset_context`… are `jsonb` and may be `NULL` on a row that never enriched (a `received` alert cannot have a triage job, but defend anyway: `None` → `{"criticality": "unknown", …}` and `lookup_status` `{"asset": "skipped", …}`, with a warning). `_Subject` mirrors `transitions._CorrelationSubject` (`alert_id, agent_name, alert_user, srcip, alert_time`) — define your own; do not import a private name.

  **7 · The fake path is the test path.** Every test passes `adapter=FakeLLM(responses=[proposer_json, verifier_json])` through `functools.partial` as the handler; the real `DeepSeekAdapter` is constructed only when `adapter is None` (production and P3-T11). `config.load()` inside the handler sees no `.env` in a worktree (DEC-047) and `LLM_MODEL_PROPOSER` unset → `llm_enabled` False → the `disabled` branch; tests that need the enabled path set `LLM_MODEL_PROPOSER=fake` with `monkeypatch.setenv` — **never** by writing a `.env`.

  **8 · Seeding.** A helper `_seed_alert(conn, **overrides)` inserts the twelve NOT-NULL-without-default columns (`alert_id, rule_id, rule_level, severity, description, agent_name, alert_time, category, resolved_by, mapping_version, event_bucket_hash, raw_payload` — P2-T07's card measured them) plus `status='queued_tier1'`, the four `*_context` jsonb columns in `finish_enrichment`'s shape, `occurrence_count`, then `enqueue(conn, "triage", alert_id)`.
