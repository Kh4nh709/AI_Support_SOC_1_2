````python
"""Extract every `psql "$DATABASE_URL" … <<'SQL'` block from docs/demo.md and run it, verbatim,
against the private test database: "$DATABASE_URL" is replaced by "$TDB" (soc_p8t04_test) and
`-v ON_ERROR_STOP=1` is added; nothing else changes. Every session is read-only (PGOPTIONS, as the
pre-flight sets it) and runs as the application role app_rw. The block that takes `AID` is run
once per fixture id given on the command line. The DSN is never printed. TDB is set by tdb.sh.
"""

from __future__ import annotations

import os
import re
import subprocess
import sys
import textwrap

DOC = "/project/project/AI_Support_SOC_1_2-P8-T04/docs/demo.md"
AIDS = sys.argv[1:] or ["1790000100.1001"]

tdb = os.environ["TDB"]
assert tdb.endswith("/soc_p8t04_test"), "refusing: not the private test database"

text = open(DOC, encoding="utf-8").read()
fences = re.findall(r"^( *)```bash\n(.*?)^\1```", text, flags=re.S | re.M)

blocks = []
for _indent, body in fences:
    lines = textwrap.dedent(body).splitlines()
    i = 0
    while i < len(lines):
        line = lines[i]
        if line.startswith('psql "$DATABASE_URL"') and line.rstrip().endswith("<<'SQL'"):
            j = i + 1
            while lines[j] != "SQL":
                j += 1
            takes_aid = any(l.startswith("AID=") for l in lines[:i])
            blocks.append((takes_aid, line, "\n".join(lines[i + 1 : j])))
            i = j + 1
        else:
            i += 1

env = dict(os.environ)
env["PGOPTIONS"] = "-c default_transaction_read_only=on -c role=app_rw"
env["COLUMNS"] = "100"  # emulate a 100-column terminal for -P format=wrapped (pre-flight line 8)


def run(script: str) -> None:
    proc = subprocess.run(["bash", "-c", script], env=env, capture_output=True, text=True)
    sys.stdout.write(proc.stdout)
    if proc.stderr:
        sys.stdout.write("[stderr] " + proc.stderr)
    print(f"----- exit {proc.returncode}")


print(f"{len(blocks)} psql heredoc blocks found in docs/demo.md")
print("\n===== pre-flight line 1 check: show default_transaction_read_only; current_user")
run('set -euo pipefail\npsql "$TDB" -XAtc "show default_transaction_read_only" -c "select current_user"')

for n, (takes_aid, cmd, sql) in enumerate(blocks, 1):
    head = cmd.replace('"$DATABASE_URL"', '"$TDB"').replace(" <<'SQL'", " -v ON_ERROR_STOP=1 <<'SQL'")
    for aid in AIDS if takes_aid else [None]:
        prelude = f"AID='{aid}'" if aid else ""
        print(f"\n===== block {n}: {cmd}" + (f"   [AID={aid}]" if aid else ""))
        run("\n".join(["set -euo pipefail", prelude, head, sql, "SQL"]))

print("\n===== the read-only guard refuses a write (nothing is written)")
run(
    'psql "$TDB" -X -v ON_ERROR_STOP=1 -c '
    "\"INSERT INTO source_heartbeat (manager_id) VALUES ('guard-probe')\" 2>&1 | tail -1"
)
````
